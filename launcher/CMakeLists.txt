cmake_minimum_required(VERSION 3.24)
project(RabbitShellPortable LANGUAGES CXX RC)
set(PORTABLE_VERSION "0.1.12" CACHE STRING "Embedded product version")
if(NOT PORTABLE_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
    message(FATAL_ERROR "Invalid portable version")
endif()
string(REPLACE "." "," PORTABLE_VERSION_COMMAS "${PORTABLE_VERSION}")
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The portable launcher requires Windows x64 and MSVC")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
set(PORTABLE_GENERATED_DIR "" CACHE PATH "Directory produced by make-portable-payload.py")
set(PORTABLE_OUTPUT_NAME "RabbitShell-${PORTABLE_VERSION}-portable" CACHE STRING "Launcher output name")
set(PORTABLE_PRODUCT_NAME "RabbitShell" CACHE STRING "Displayed product name")
if(NOT PORTABLE_PRODUCT_NAME MATCHES "^(RabbitShell|FrogShell)$")
    message(FATAL_ERROR "Unsupported product name")
endif()
foreach(required payload.bin payload_meta.h RabbitShell.ico)
    if(NOT EXISTS "${PORTABLE_GENERATED_DIR}/${required}")
        message(FATAL_ERROR "Missing generated input: ${required}")
    endif()
endforeach()
configure_file(launcher.rc.in "${CMAKE_CURRENT_BINARY_DIR}/launcher.rc" @ONLY)
add_executable(RabbitShellPortable WIN32 launcher.cpp "${CMAKE_CURRENT_BINARY_DIR}/launcher.rc")
target_compile_features(RabbitShellPortable PRIVATE cxx_std_20)
target_include_directories(RabbitShellPortable PRIVATE "${PORTABLE_GENERATED_DIR}")
target_compile_definitions(RabbitShellPortable PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 RABBITSHELL_PORTABLE_VERSION="${PORTABLE_VERSION}")
target_compile_options(RabbitShellPortable PRIVATE /W4 /WX /utf-8 /guard:cf /Brepro /Zc:__cplusplus)
target_link_options(RabbitShellPortable PRIVATE /INCREMENTAL:NO /Brepro /DEBUG:NONE /guard:cf /DYNAMICBASE /HIGHENTROPYVA /NXCOMPAT /DEPENDENTLOADFLAG:0x800 /MANIFEST:NO)
set_target_properties(RabbitShellPortable PROPERTIES OUTPUT_NAME "${PORTABLE_OUTPUT_NAME}")
add_executable(portable-child-probe EXCLUDE_FROM_ALL tests/child_probe.cpp)
target_compile_features(portable-child-probe PRIVATE cxx_std_20)
target_compile_definitions(portable-child-probe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(portable-child-probe PRIVATE /W4 /WX /utf-8 /Brepro)
target_link_options(portable-child-probe PRIVATE /INCREMENTAL:NO /Brepro /DEBUG:NONE /DEPENDENTLOADFLAG:0x800)
add_executable(portable-origin-probe EXCLUDE_FROM_ALL tests/origin_probe.cpp)
target_compile_features(portable-origin-probe PRIVATE cxx_std_20)
target_compile_definitions(portable-origin-probe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(portable-origin-probe PRIVATE /W4 /WX /utf-8 /Brepro)
target_link_options(portable-origin-probe PRIVATE /INCREMENTAL:NO /Brepro /DEBUG:NONE /DEPENDENTLOADFLAG:0x800)
